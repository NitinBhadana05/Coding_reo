#Consecutive Numbers Check

def check(numbers: list[int]) -> bool:
    
    if not numbers:
        return False

    min_val = numbers[0]
    max_val = numbers[0]

    for num in numbers:
        if num < min_val:
            min_val = num
        if num > max_val:
            max_val = num

    if max_val - min_val + 1 != len(numbers):
        return False

    for i in range(len(numbers)):
        for j in range(i + 1, len(numbers)):
            if numbers[i] == numbers[j]:
                return False

    return True
    
#example

numbers = [1, 2, 3, 5]
print(check(numbers))

#output: 
#False
